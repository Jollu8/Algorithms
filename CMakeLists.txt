cmake_minimum_required(VERSION 3.27)
project(algorithms LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(SANITIZE "Enable AddressSanitizer" OFF)
set(ALGORITHM_SOURCE "" CACHE FILEPATH "Standalone C++ source to build as selected_algorithm")

function(add_algorithm name source)
    add_executable(${name} "${source}")
    if(SANITIZE)
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(${name} PRIVATE -fsanitize=address -fno-omit-frame-pointer)
            target_link_options(${name} PRIVATE -fsanitize=address)
        else()
            message(FATAL_ERROR "SANITIZE requires GCC or Clang")
        endif()
    endif()
endfunction()

# Explicit targets: the collection also contains unfinished exercises.
add_algorithm(sorting_bubble algorithms/sorting/Bubble.cpp)
add_algorithm(sorting_counting algorithms/sorting/Counting.cpp)
add_algorithm(sorting_insertion algorithms/sorting/Insertion.cpp)
add_algorithm(sorting_merge algorithms/sorting/Merge.cpp)
add_algorithm(sorting_quick algorithms/sorting/Quick.cpp)
add_algorithm(sorting_quick_recursive algorithms/sorting/QuicSortRecursive.cpp)
add_algorithm(sorting_selection algorithms/sorting/Selection.cpp)
add_algorithm(sorting_shell algorithms/sorting/Shell.cpp)
add_algorithm(sorting_stupid algorithms/sorting/Stupid.cpp)

if(ALGORITHM_SOURCE)
    get_filename_component(selected_source "${ALGORITHM_SOURCE}" ABSOLUTE BASE_DIR "${PROJECT_SOURCE_DIR}")
    if(NOT EXISTS "${selected_source}" OR IS_DIRECTORY "${selected_source}")
        message(FATAL_ERROR "ALGORITHM_SOURCE must point to an existing C++ source: ${selected_source}")
    endif()
    find_package(Threads REQUIRED)
    add_algorithm(selected_algorithm "${selected_source}")
    target_link_libraries(selected_algorithm PRIVATE Threads::Threads)
endif()
